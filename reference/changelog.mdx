export const openEntry = (el, data) => {
  document.querySelectorAll('.cl-entry').forEach(e => e.classList.remove('cl-active'));
  el.classList.add('cl-active');
  window._clEntry = el;

  const win = document.getElementById('cl-window');

  // Position horizontally in the whitespace to the right of the content column
  const contentArea = document.getElementById('content-area');
  if (contentArea) {
    const cRect = contentArea.getBoundingClientRect();
    win.style.left = (cRect.right + 12) + 'px';
    win.style.width = Math.min(300, window.innerWidth - cRect.right - 24) + 'px';
  } else {
    win.style.right = '16px';
    win.style.width = '280px';
  }

  // Align vertically with the clicked entry
  const eRect = el.getBoundingClientRect();
  win.style.top = Math.max(80, Math.min(eRect.top, window.innerHeight - 260)) + 'px';
  win.style.display = 'block';

  // Attach scroll listener once; rAF-throttled so it runs at most once per paint frame
  if (!window._clScrollBound) {
    document.addEventListener('scroll', () => {
      if (window._clRaf) return;
      window._clRaf = requestAnimationFrame(() => {
        window._clRaf = null;
        const entry = window._clEntry;
        const w = document.getElementById('cl-window');
        if (!entry || !w || w.style.display === 'none') return;
        const r = entry.getBoundingClientRect();
        w.style.top = Math.max(80, Math.min(r.top, window.innerHeight - 260)) + 'px';
      });
    }, { passive: true, capture: true });
    window._clScrollBound = true;
  }

  document.getElementById('cl-version').textContent = data.version;
  document.getElementById('cl-date').textContent = data.date;
  document.getElementById('cl-title').textContent = data.title;
  document.getElementById('cl-desc').textContent = data.desc;
  const linkEl = document.getElementById('cl-link');
  if (data.link) {
    linkEl.href = data.link;
    document.getElementById('cl-link-label').textContent = data.linkLabel;
    linkEl.style.display = 'inline-flex';
  } else {
    linkEl.style.display = 'none';
  }
};

  document.querySelectorAll('.cl-entry').forEach(e => e.classList.remove('cl-active'));
};
