export const openEntry = (el, data) => {
  document.querySelectorAll('.cl-entry').forEach(e => e.classList.remove('cl-active'));
  el.classList.add('cl-active');
  window._clEntry = el;
  window._clLinkTarget = data.link || null;

  // Build portal once, appended to <body> so it's outside any Mintlify stacking context
  if (!document.getElementById('cl-portal')) {
    const isDark = document.documentElement.classList.contains('dark');
    const p = document.createElement('div');
    p.id = 'cl-portal';
    p.style.cssText = 'position:fixed;z-index:99999;display:none;';
    p.innerHTML = `
      <div style="border-radius:12px;border:1px solid rgba(${isDark?'255,255,255,0.1':'0,0,0,0.08'});background:${isDark?'#111827':'#fff'};box-shadow:0 4px 24px rgba(0,0,0,${isDark?'0.4':'0.1'});overflow:hidden;">
        <div style="display:flex;align-items:center;justify-content:space-between;padding:10px 14px;border-bottom:1px solid rgba(${isDark?'255,255,255,0.08':'0,0,0,0.08'});background:${isDark?'rgba(31,41,55,0.5)':'#f9fafb'};">
          <span id="clp-ver" style="font-family:ui-monospace,monospace;font-size:11px;font-weight:600;color:#ea580c;"></span>
          <button onclick="window._clClose()" style="width:20px;height:20px;border-radius:50%;background:${isDark?'#374151':'#e5e7eb'};border:none;cursor:pointer;font-size:10px;color:${isDark?'#9ca3af':'#6b7280'};line-height:1;flex-shrink:0;">✕</button>
        </div>
        <div style="padding:16px;display:flex;flex-direction:column;gap:10px;">
          <span id="clp-date" style="font-family:ui-monospace,monospace;font-size:10px;text-transform:uppercase;letter-spacing:0.08em;color:#9ca3af;"></span>
          <div id="clp-title" style="font-family:'Instrument Serif',Georgia,serif;font-size:14px;font-style:italic;color:${isDark?'#f9fafb':'#111827'};line-height:1.4;"></div>
          <p id="clp-desc" style="font-size:12px;color:${isDark?'#9ca3af':'#6b7280'};line-height:1.6;margin:0;"></p>
          <button id="clp-link" onclick="if(window._clLinkTarget)window.location.href=window._clLinkTarget" style="display:none;font-family:ui-monospace,monospace;font-size:12px;color:#ea580c;background:none;border:none;padding:0;cursor:pointer;text-align:left;margin-top:2px;"></button>
        </div>
      </div>`;
    document.body.appendChild(p);

    window._clClose = () => {
      p.style.display = 'none';
      document.querySelectorAll('.cl-entry').forEach(e => e.classList.remove('cl-active'));
      window._clEntry = null;
    };

    // rAF-throttled scroll tracking
    document.addEventListener('scroll', () => {
      if (window._clRaf) return;
      window._clRaf = requestAnimationFrame(() => {
        window._clRaf = null;
        if (!window._clEntry || p.style.display === 'none') return;
        const r = window._clEntry.getBoundingClientRect();
        p.style.top = Math.max(80, Math.min(r.top, window.innerHeight - 280)) + 'px';
      });
    }, { passive: true, capture: true });
  }

  const portal = document.getElementById('cl-portal');

  // Horizontal: right of content column
  const ca = document.getElementById('content-area');
  if (ca) {
    const cr = ca.getBoundingClientRect();
    portal.style.left = (cr.right + 12) + 'px';
    portal.style.width = Math.min(300, window.innerWidth - cr.right - 24) + 'px';
  } else {
    portal.style.right = '16px';
    portal.style.width = '280px';
  }
  // Vertical: aligned with clicked entry
  const er = el.getBoundingClientRect();
  portal.style.top = Math.max(80, Math.min(er.top, window.innerHeight - 280)) + 'px';
  portal.style.display = 'block';

  // Populate
  document.getElementById('clp-ver').textContent = data.version;
  document.getElementById('clp-date').textContent = data.date;
  document.getElementById('clp-title').textContent = data.title;
  document.getElementById('clp-desc').textContent = data.desc;
  const lnk = document.getElementById('clp-link');
  if (data.link) { lnk.textContent = data.linkLabel + ' →'; lnk.style.display = 'block'; }
  else { lnk.style.display = 'none'; }
};
